name: Schema Diff
on:
  pull_request:
    paths:
      - '**/*.sql'
      - 'db/schema.sql'
      - 'migrations/**'
jobs:
  schema-diff:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: aimadetools/race-kimi@main
        with:
          schema_path: 'db/schema.sql'
